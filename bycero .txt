local function sendDiscordReport(targetPlayer, reason)
    if not targetPlayer then return end
    
    local currentTime = os.date("*t")
    local fecha = string.format("%02d/%02d/%04d", currentTime.day, currentTime.month, currentTime.year)
    local hora = string.format("%02d:%02d:%02d", currentTime.hour, currentTime.min, currentTime.sec)
    
    local dias = {"Domingo", "Lunes", "Martes", "Miércoles", "Jueves", "Viernes", "Sábado"}
    local diaSemana = dias[currentTime.wday] or "Desconocido"

    local serverLink = "https://www.roblox.com/games/" .. tostring(game.PlaceId) .. "?jobId=" .. tostring(game.JobId)

    local accountAgeDays = targetPlayer.AccountAge or 0
    local creationTimestamp = os.time() - (accountAgeDays * 86400)
    local creationDate = os.date("%d/%m/%Y", creationTimestamp)
    
    local totalYears = math.floor(accountAgeDays / 365)
    local remainingDaysAfterYears = accountAgeDays % 365
    local totalMonths = math.floor(remainingDaysAfterYears / 30)
    local remainingDays = remainingDaysAfterYears % 30

    local formattedAge = ""
    if totalYears > 0 then formattedAge = formattedAge .. totalYears .. (totalYears == 1 and " año, " or " años, ") end
    if totalMonths > 0 then formattedAge = formattedAge .. totalMonths .. (totalMonths == 1 and " mes, " or " meses, ") end
    formattedAge = formattedAge .. remainingDays .. (remainingDays == 1 and " día" or " días") .. " (" .. tostring(accountAgeDays) .. " días en total)"

    local payload = {
        ["embeds"] = {{
            ["title"] = "🚨 Nuevo Reporte de Staff - Chamos city",
            ["color"] = 16724787,
            ["fields"] = {
                { ["name"] = "👤 Usuario Reportado", ["value"] = "**Display:** " .. targetPlayer.DisplayName .. "\n**Username:** @" .. targetPlayer.Name .. "\n**ID:** " .. tostring(targetPlayer.UserId), ["inline"] = false },
                { ["name"] = "📅 Creación de la Cuenta", ["value"] = "**Fecha:** " .. creationDate, ["inline"] = true },
                { ["name"] = "⏳ Tiempo de la Cuenta", ["value"] = formattedAge, ["inline"] = true },
                { ["name"] = "📝 Motivo / Razón", ["value"] = "```" .. reason .. "```", ["inline"] = false },
                { ["name"] = "🛡️ Staff Emisor", ["value"] = "**Display:** " .. LocalPlayer.DisplayName .. "\n**Username:** @" .. LocalPlayer.Name .. "\n**ID:** " .. tostring(LocalPlayer.UserId), ["inline"] = false },
                { ["name"] = "🎮 Servidor / Juego", ["value"] = "[Chamos city] (" .. serverLink .. ")", ["inline"] = false },
                { ["name"] = "📅 Fecha del Reporte", ["value"] = diaSemana .. " - " .. fecha, ["inline"] = true },
                { ["name"] = "⏰ Hora", ["value"] = hora, ["inline"] = true }
            },
            ["footer"] = { ["text"] = "Sistema de Reportes Chamos City Staff" }
        }}
    }

    local requestFunc = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
    if requestFunc then
        sendReportModalBtn.Text = "ENVIANDO..."
        sendReportModalBtn.BackgroundColor3 = Color3.fromRGB(120, 120, 120)

        task.spawn(function()
            requestFunc({
                Url = WEBHOOK_URL,
                Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = HttpService:JSONEncode(payload)
            })
            
            sendReportModalBtn.Text = "¡ENVIADO CON ÉXITO! ✓"
            sendReportModalBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 80)
            
            task.delay(1.2, function()
                reportModalFrame.Visible = false
                reportReasonBox.Text = ""
                sendReportModalBtn.Text = "ENVIAR REPORTE"
                sendReportModalBtn.BackgroundColor3 = Color3.fromRGB(210, 80, 0)
            end)
        end)
    end
end

local function stopSpectating()
    spectatingTarget = nil
    spectateBar.Visible = false
    reportModalFrame.Visible = false
    
    if LocalPlayer.Character then
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum then
            Camera.CameraSubject = hum
        else
            Camera.CameraSubject = LocalPlayer.Character
        end
    end
end

local function setSpectateTarget(plr)
    if not plr or not plr.Character then return end
    
    local targetSubject = plr.Character:FindFirstChildOfClass("Humanoid") or plr.Character:FindFirstChild("HumanoidRootPart")
    if targetSubject then
        Camera.CameraSubject = targetSubject
        spectatingTarget = plr
        
        specNameLabel.Text = plr.DisplayName
        currentUsernameToCopy = plr.Name
        userCopyBtn.Text = "@" .. plr.Name
        userCopyBtn.TextColor3 = Color3.fromRGB(0, 200, 255)

        pcall(function()
            specAvatar.Image = Players:GetUserThumbnailAsync(plr.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
        end)
        spectateBar.Visible = true
    end
end

local function updateList()
    for _, child in ipairs(scroll:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end

    local playersList = Players:GetPlayers()
    onlineLabel.Text = "Online: " .. #playersList
    local filterText = string.lower(searchBox.Text)

    for _, plr in ipairs(playersList) do
        if filterText == "" or string.find(string.lower(plr.DisplayName), filterText) or string.find(string.lower(plr.Name), filterText) then
            createPlayerCard(plr)
        end
    end
end
